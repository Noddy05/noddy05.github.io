2) Recursive backtracker

RecursiveBacktracker(maze, startingCell)
    path = [ startingCell ]
    maze.visit(startingCell)
    
    while len(path) > 0
        //Let pathEnd be the last element in the path array.
        pathEnd = path.last()
        neighbours = maze.unvisitedNeighbours(pathEnd)
        
        if len(neighbours) == 0
            path.pop()
        else
            pickedNeighbour = neighbours[floor(random(0, len(neighbours)))]
            maze.connectCells(pathEnd, pickedNeighbour)
            path.append(pickedNeighbour)
        endif
    endwhile
