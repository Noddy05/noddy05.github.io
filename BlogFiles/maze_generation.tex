\section{Maze Generation Algorithms}

\subsection{Recursive backtracker}
\div{recursive_backtracker}{algorithm_window}
\code{##RecursiveBacktracker##(~maze~, ~startingCell~)
    path = [ ~startingCell~ ]
    ~maze~.##visit##(~startingCell~)
    
    while len(path) > 0
        //Let pathEnd be the last element in the path array.
        pathEnd = path.##last##()
        neighbours = ~maze~.##unvisitedNeighbours##(pathEnd)
        
        if len(neighbours) == 0
            path.##pop##()
        else
            pickedNeighbour = neighbours[##random##(0, len(neighbours))]
            ~maze~.##connectCells##(pathEnd, pickedNeighbour)
            path.##append##(pickedNeighbour)
        endif
    endwhile
}

\subsection{Bacterial growth}
\div{bacterial_growth}{algorithm_window}
\code{##BacterialGrowth##(~maze~, ~startingCell~)
    bacteria = [ ~startingCell~ ]
    ~maze~.##visit##(~startingCell~)
    
    while len(bacteria) > 0
        //As we loop through the array some bacteria will not have any unvisited
        //neighbours. We mark their indices and remove them afterwards.
        deadBacteria = []

        for i = 0 to len(bacteria)
            neighbours = ~maze~.##unvisitedNeighbours##(bacteria[i])
            
            //If a bacteria cant spead, we mark it as dead
            if len(neighbours) == 0
                deadBacteria.##append##(i)
            else
                //Otherwise we let it spread to a random neighbour
                pickedNeighbour = neighbours[##random##(0, len(neighbours))]
                ~maze~.##connectCells##(pathEnd, pickedNeighbour)
                bacteria.##append##(pickedNeighbour)
            endif
        endfor

        //Finally remove the bacteria, and remember that after we removed the
        //first bacteria, the indices will be off by 1, and off by 2 after removing
        //the second, etc... therefore we subtract the index by i
        for i = 0 to len(deadBacteria)
            bacteria.##removeAt##(deadBacteria[i] - i)
        endfor
    endwhile
}
